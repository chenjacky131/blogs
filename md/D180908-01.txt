对象的深拷贝
----------

by jacky at 2018/12/24
````
/**
 * @description: 对象的深拷贝
 * @param {*} obj 要拷贝的对象
 * @param {*} map  用于存储对象循环引用的地址
 * @return {*}
 */
function deep(obj = {}, map = new Map()){
  if(typeof obj !== 'object'){  //  基本数据类型
    return obj;
  }
  if(map.get(obj)){ //  循环引用的对象
    return map.get(obj);
  }
  if(Object.prototype.toString.call(obj).slice(8, -1) === 'Date'){ //  Date特殊对象
    return new Date(obj);
  }
  if(Object.prototype.toString.call(obj).slice(8, -1) === 'RegExp'){ //  RegExp特殊对象
    return new RegExp(obj);
  }
  let result = {}
  //  初始化放回结果
  if(obj instanceof Array ||
  //  加||是为了防止Array的prototype被重写， Array.isArray也是如此。
  Object.prototype.toString(obj) === '[object Array]'){
    result = []
  }
  //  防止循环引用
  map.set(obj, result);
  for(let key in obj){
    //  保证key不是原型属性
    if(obj.hasOwnProperty(key)){
      //  递归调用
      result[key] = deep(obj[key], map);
    }
  }
  return result;
}
````
