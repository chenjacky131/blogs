mysql的一些命令
----------

by jack at 2022/09/02

````
// 查看mysql安装目录
which mysqld

// 查看mysql配置文件
/usr/sbin/mysqld(上一步查出来的安装目录) --verbose --help |grep -A 1 'Default options'

// 重启mysql
service mysqld restart

// 查看mysql状态
service mysqld status

//  替换表字段名
update 表名 set 字段名 = replace(字段名,'旧网址','新网址')
````
